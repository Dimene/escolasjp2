    jQuery(document).ready(function($) {
        // Inicialização
        $reverterpermisao=@json($reverterpermisao);

        if($reverterpermisao==false){
$(".remover_elemento").prop("disabled", true);
        }
        updateMonthDisplay($('.select-mes').val());

        // Atualiza a exibição quando o mês é alterado
        $(".select-mes").change(function() {
            var selectedMonth = $(this).val();
            $("input[name='mesid']").val(selectedMonth);
            updateMonthDisplay(selectedMonth);
        });

        // Alterna entre formulário normal e MPesa
        $(".tipopagamento").change(function() {
            var paymentMethod = $(this).val();
            var currentMonth = $('.select-mes').val();

            if(paymentMethod === "Mpesa-Online") {
                $(".month-container").hide();
                $("#month-" + currentMonth + " .mpesa-form").show();
            } else {
                updateMonthDisplay(currentMonth);
            }
        });


        // Estilização dos botões de pagamento
        $(document).on('click', '.payment-options .btn', function() {
            var parent = $(this).closest('.btn-group');
            parent.find('.btn').removeClass('btn-success active').addClass('btn-outline-secondary');
            $(this).removeClass('btn-outline-secondary').addClass('btn-success active');

            // Atualiza o status no badge
            var status = $(this).find('input').val() === "Pago" ? "Pago" : "Pendente";
            var monthContainer = $(this).closest('.month-container');
            var badge = monthContainer.find('.month-status-badge .badge');

            badge.removeClass('badge-success badge-warning badge-danger');

            if(status === "Pago") {
                badge.addClass('badge-success').text('Pago');
            } else {
                var dataLimite = new Date(monthContainer.find('.month-details small:first').text().replace('Prazo: ', ''));
                var hoje = new Date();

                if(hoje > dataLimite) {
                    badge.addClass('badge-danger').text('Atrasado');
                } else {
                    badge.addClass('badge-warning').text('Pendente');
                }
            }
        });

        // Envio do formulário
        $(document).on('click',".Guardar-mensalida", function(e) {
            e.preventDefault();

            var submitBtn = $(this).find('.Guardar-mensalida');
            submitBtn.prop('disabled', true);
            submitBtn.find('.submit-text').addClass('d-none');
            submitBtn.find('.spinner-border').removeClass('d-none');


           var estado = [];
estado[0] = $("[name='options" + $("[name='mesid']").val() + "']:checked").val();

var mes = [];
mes[0] = $("[name='mesid']").val();


var multaactiva = [];
multaactiva[0] =0;
if($("[name='multaactiva" + $("[name='mesid']").val() + "']").is(':checked')){
multaactiva[0] = $("[name='multaactiva" + $("[name='mesid']").val() + "']").val();
}


            var formData = {
                "idclass": $("[name='idclass']").val(),
                "flag": 1,
                "mesid":mes ,
                "tipopagamento": $(".tipopagamento").val(),
                "tipopagamentoapagar": $("[name='tipopagamentoapagar']").val(),
                "multaactiva": multaactiva,
                "numero": $("[name='numero" + $("[name='mesid']").val() + "']").val(),
                "estado": estado
            };

            if(parseInt(formData.flag) === 2) {
                formData.estado = $("[name='options1']:checked").val();
            }

            console.log("Dados do formulário:", formData);

            $.ajax({
                url: "/RegistoAcademico/outrosPagamento/atualizardados/" + formData.idclass,
                type: 'GET',
                data: formData,
                success: function(data) {
                    if (data == 0) {
                        $(".inforeversao").html(
                            '<div class="alert alert-success">Reversão efetuada com sucesso!</div>'
                        );
                        setTimeout(function() {
                            $(".inforeversao").fadeOut();
                        }, 3000);
                    } else {
                        var tela_impressao = window.open('about:blank');
                        tela_impressao.document.write(data);
                        tela_impressao.window.print();
                        tela_impressao.window.close();
                    }
                },
                error: function() {
                    $(".inforeversao").html(
                        '<div class="alert alert-danger">Erro ao processar a requisição!</div>'
                    );
                },
                complete: function() {
                    submitBtn.prop('disabled', false);
                    submitBtn.find('.submit-text').removeClass('d-none');
                    submitBtn.find('.spinner-border').addClass('d-none');
                }
            });
        });
    });

        function updateMonthDisplay(monthId) {
            // Oculta todos os meses
            $(".month-container").hide();

            // Mostra apenas o mês selecionado

            // Atualiza o formulário MPesa conforme necessário
